<script>
  // Mobile menu toggle
  const btnMenu = document.getElementById('btnMenu') || document.getElementById('btnMenuMobile');
  const mobileNav = document.getElementById('mobileNav');
  if(btnMenu){
    btnMenu.addEventListener('click',()=> mobileNav.classList.toggle('hidden'));
  }

  // Produtos (demo data)
  const products = [
    { name: 'Câmera IP 4K', price: 1200, category: 'Câmeras', image: 'https://images.unsplash.com/photo-1581093588401-6a2b7d207f49?auto=format&fit=crop&w=800&q=80', tags: ['ip','4k'] },
    { name: 'DVR 8 canais', price: 850, category: 'Gravação', image: 'https://images.unsplash.com/photo-1611175694116-3f0e43c9817b?auto=format&fit=crop&w=800&q=80', tags: ['dvr','8 canais'] },
    { name: 'Alarme Residencial', price: 450, category: 'Alarmes', image: 'https://images.unsplash.com/photo-1600267166790-fd5c3f9ffbf7?auto=format&fit=crop&w=800&q=80', tags: ['alarme','residencial'] },
    { name: 'Cerca Elétrica 20m', price: 300, category: 'Cerca', image: 'https://images.unsplash.com/photo-1618889174733-2c33c7f99945?auto=format&fit=crop&w=800&q=80', tags: ['cerca','elétrica'] },
    { name: 'Controle de Acesso', price: 950, category: 'Acesso', image: 'https://images.unsplash.com/photo-1612831455542-0d69f4ed5b8e?auto=format&fit=crop&w=800&q=80', tags: ['acesso','controle'] },
    { name: 'Cabo Coaxial 100m', price: 120, category: 'Acessórios', image: 'https://images.unsplash.com/photo-1580584126364-5b5efc37d0a5?auto=format&fit=crop&w=800&q=80', tags: ['cabo','coaxial'] }
  ];

  const grid = document.getElementById('grid');
  const emptyState = document.getElementById('emptyState');
  const searchInput = document.getElementById('search');
  const categorySelect = document.getElementById('category');
  const sortSelect = document.getElementById('sort');

  function formatPrice(v){ return v.toLocaleString('pt-BR',{minimumFractionDigits:2}); }

  function renderProducts(){
    const search = (searchInput?.value || '').toLowerCase();
    const category = categorySelect?.value || '';
    let filtered = products.filter(p=> p.name.toLowerCase().includes(search) || p.tags.some(t=> t.includes(search)));
    if(category) filtered = filtered.filter(p=> p.category === category);
    if(sortSelect){
      if(sortSelect.value === 'asc') filtered.sort((a,b)=>a.price-b.price);
      else if(sortSelect.value === 'desc') filtered.sort((a,b)=>b.price-a.price);
    }

    grid.innerHTML = '';
    if(filtered.length===0){ emptyState.classList.remove('hidden'); }
    else{
      emptyState.classList.add('hidden');
      filtered.forEach(p=>{
        const card = document.createElement('article');
        card.className = 'bg-neutral-900 border border-neutral-800 rounded-2xl overflow-hidden shadow hover:shadow-2xl transition transform hover:-translate-y-1';
        card.innerHTML = `
          <img src="${p.image}" alt="${p.name}" class="w-full h-44 object-cover" loading="lazy"/>
          <div class="p-4">
            <h3 class="font-semibold text-white">${p.name}</h3>
            <p class="text-neutral-400 text-sm mt-1">${p.category}</p>
            <div class="mt-4 flex items-center justify-between">
              <div class="text-emerald-400 font-bold">R$ ${formatPrice(p.price)}</div>
              <a href="https://wa.me/5544991561510?text=Quero%20um%20or%C3%A7amento%20do%20${encodeURIComponent(p.name)}" target="_blank" class="inline-flex items-center rounded-full bg-emerald-600 text-black px-3 py-1 text-sm font-semibold">Orçar</a>
            </div>
          </div>
        `;
        grid.appendChild(card);
      });
    }
  }

  searchInput?.addEventListener('input', renderProducts);
  categorySelect?.addEventListener('change', renderProducts);
  sortSelect?.addEventListener('change', renderProducts);
  renderProducts();
</script>
