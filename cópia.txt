    const grid = document.getElementById('grid');
    const emptyState = document.getElementById('emptyState');
    const search = document.getElementById('search');
    const category = document.getElementById('category');
    const sort = document.getElementById('sort');

    function renderProducts(list) {
      grid.innerHTML = list.map(p => `
        <div class="rounded-2xl border border-neutral-200 bg-white overflow-hidden shadow-sm hover:shadow-lg transition">
          <img src="${p.image}" alt="${p.name}" class="w-full h-48 object-contain p-4 bg-white" loading="lazy" />

            <p class="mt-2 font-bold text-sky-600">R$ ${p.price.toFixed(2)}</p>
          </div>
        </div>
      `).join('');
      emptyState.classList.toggle('hidden', list.length > 0);
    }

    function updateProducts() {
      const term = search.value.toLowerCase();
      const cat = category.value;
      const sortVal = sort.value;

      let filtered = products.filter(p =>
        (!cat || p.category === cat) &&
        (p.name.toLowerCase().includes(term) || p.tags.some(tag => tag.includes(term)))
      );

      if (sortVal === 'asc') filtered.sort((a,b)=>a.price-b.price);
      if (sortVal === 'desc') filtered.sort((a,b)=>b.price-a.price);
      if (sortVal === 'pop') filtered.sort((a,b)=>b.popular - a.popular);

      renderProducts(filtered);
    }

    [search, category, sort].forEach(el => el.addEventListener('input', updateProducts));
    updateProducts();
